cmake_minimum_required(VERSION 3.22)
project(asterion_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Static llama/ggml archives are linked into this JNI shared library.
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
if(NOT DEFINED LLAMA_SOURCE_DIR OR NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
    message(FATAL_ERROR "Pinned llama.cpp source checkout required")
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
# The pinned Vulkan backend finds SPIRV-Headers but does not propagate
# its include directory to all of the backend's compilation units.
if(GGML_VULKAN)
    if(NOT TARGET ggml-vulkan)
        message(FATAL_ERROR "Vulkan requested but native ggml-vulkan target missing")
    endif()
    find_package(SPIRV-Headers CONFIG REQUIRED)
    target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
endif()
if(NOT TARGET mtmd)
    message(FATAL_ERROR "Pinned native multimodal library missing")
endif()
add_library(asterion_vision_core STATIC native_vision.cpp)
target_link_libraries(asterion_vision_core PUBLIC llama mtmd)
add_library(asterion_llm SHARED on_device_llm.cpp)
target_compile_definitions(asterion_llm PRIVATE ASTERION_VISION=1)
target_link_libraries(asterion_llm PRIVATE asterion_vision_core)
option(ASTERION_BUILD_VISION_PROBE "Build host qualification executable" OFF)
if(ASTERION_BUILD_VISION_PROBE AND NOT ANDROID)
    add_executable(asterion_vision_probe "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../tests/native/vision_probe.cpp")
    target_include_directories(asterion_vision_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(asterion_vision_probe PRIVATE asterion_vision_core)
endif()
target_include_directories(asterion_llm PRIVATE "${LLAMA_SOURCE_DIR}/include")
if(ANDROID)
    target_link_libraries(asterion_llm PRIVATE llama android log)
else()
    find_package(JNI REQUIRED)
    target_include_directories(asterion_llm PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(asterion_llm PRIVATE llama)
endif()
